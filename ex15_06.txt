// p5.js: 음성 제어로 이동하는 원의 x축 위치좌표값 송신, 가변저항값 수신하여 원의 크기 변화

let speech = new p5.Speech();
let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let result = '', value1 = 0, value2 = 0, x = 200, dx = 0;

function setup() {
  createCanvas(400, 400);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 오른쪽, 왼쪽, 정지', 20, 30);
  text('LED 밝기: ' + nf(value1, 3), 20, 60);
  text('가변저항: ' + nf(value2, 3), 20, 90);
  
  if (result.indexOf('오른쪽')!==-1) {
    dx = 1;
  } else if (result.indexOf('왼쪽')!==-1) {
    dx = -1;
  } else if (result.indexOf('정지')!==-1) {
    dx = 0;
  }
  x += dx;
  x = constrain(x, 0, width);
  line(x, 0, x, height);

  noFill();
  stroke(255, 255, 0);
  ellipse(width/2, height/2, value2, value2);
}

function mousePressed() {
  speech.speak(`밝기값은 ${value1}이고 가변저항값은 ${value2}입니다`);
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write(0);
}

function serialEvent() {
  value2 = port.read();
  value2 = int(map(value2, 0, 255, 0, width-1));
  value1 = int(map(x, 0, width, 0, 255));
  port.write(value1);
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
