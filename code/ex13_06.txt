// p5.js: 음성 제어(→왼쪽, 오른쪽, 위쪽, 아래쪽, 정지)로 원의 x축 및 y축 위치값(0∼255) 송신

let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let result = '';
let value1 = 0, value2 = 0;
let x = 200, y = 100, dx = 0, dy = 0;

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 왼쪽, 오른쪽, 위쪽, 아래쪽, 정지', 20, 30);
  text('Angle: ' + nf(value1, 3), 20, 60);
  text('Brightness: ' + nf(value2, 3), 20, 90);
  
  if (result.indexOf('왼쪽') !== -1) {
    dx = -1;
  } else if (result.indexOf('오른쪽') !== -1) {
    dx = 1;
  } else if (result.indexOf('위쪽') !== -1) {
    dy = -1;
  } else if (result.indexOf('아래쪽') !== -1) {
    dy = 1;
  } else if (result.indexOf('정지') !== -1) {
    dx = 0;
    dy = 0;
  }
  x += dx;
  y += dy;
  x = constrain(x, 0, width);
  y = constrain(y, 0, height);
  fill(255, 0, 0, 200);
  ellipse(x, y, 50, 50);
  
  value1 = int(map(x, 0, width, 0, 180)); 
  value2 = int(map(y, 0, height, 0, 255));
  port.write('A');
  port.write(value1);
  port.write(value2);
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
