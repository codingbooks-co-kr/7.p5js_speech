// p5.js: 버튼값 수신하여 원 색상을 변경하고 합성음성 출력

let speech = new p5.Speech();
let port = new p5.SerialPort();
let value, prevValue;

function setup() {
  createCanvas(400, 400);
  port.open('COM5');
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
}

function draw() {
  background(0);
  if (value === 0) {
    fill(255, 255, 0);
    if (prevValue === 1) {
      speech.speak('버튼을 눌렀어요');
    }
  } else if (value === 1) {
    fill(255, 0, 0);
  }
  ellipse(width/2, height/2, 200, 200);
  prevValue = value;
}

function serialEvent() {
  value = port.read();
}
