// 음성인식 결과와 신뢰도를 캔버스상에 표시

let rec = new p5.SpeechRec('ko-KR', recResult);
let result = 'None', confidence = 0;

function setup() {
  createCanvas(400, 200);
  rec.onEnd = recEnd;
  rec.start(true, false);
}

function draw() {
  background(0);
  fill(255);
  textSize(20);
  text('[음성인식 시작]', 20, 40);

  fill(255, 255, 0);
  text('인식 결과: ' + result, 20, 100);
  text('인식신뢰도: ' + nf(confidence, 1, 2), 20, 150);
}

function recResult() {
  result = rec.resultString;
  confidence = rec.resultConfidence;
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
