// p5.js: 조도센서값을 수신하여 텍스트를 상하 이동하고 밝기 정도를 음성으로 출력

let speech = new p5.Speech();
let port = new p5.SerialPort();
let value = 0, newValue = 0, num = 0, prevNum = 0;
let MIN = 400, MAX = 800;

function setup() {
  createCanvas(300, 500);
  port.open('COM5');
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
}

function draw() {
  background(0);
  stroke(255);
  line(0, height/3, width, height/3);
  line(0, height*2/3, width, height*2/3);
  
  fill(255, 255, 0);
  textSize(50);
  textAlign(CENTER);
  text(nf(value, 3), width/2, newValue);
}

function mousePressed() {
  speech.speak(`조도센서값은 ${value}입니다`);
}

function serialEvent() {
  let str = port.readLine();
  if (!str) return;
  trim(str);
  value = int(str);
  newValue = map(value, MIN, MAX, 0, height-1);
  
  num = int(newValue / (height/3));
  if (num != prevNum) {
    if (num === 0) {
      speech.speak('어두워요');
    } else if (num === 1) {
      speech.speak('밝아요');
    } else if (num === 2) {
      speech.speak('매우 밝아요');
    }
  }
  prevNum = num;
}
