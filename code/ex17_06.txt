// 음성 제어(→정지/중지)로 동물 이모지 및 LED 각 한 개 랜덤 선택

let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let index = 0, stop = false, result = '';
let animals = ['🐵', '🐱', '🐯', '🐷', '🐪', '🐘'];

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  textAlign(CENTER, CENTER);
  text('인식단어: 정지(중지)', width/2, 30);
  textSize(100);
  text(animals[index], width/2, height/2);
  
  if (stop) {
    noLoop();
  } else {
    index = int(random(animals.length));
  }
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write('A');
}

function serialEvent() {
  let signal = port.read(); 
  if (result.indexOf('정지') !== -1 || result.indexOf('중지') !== -1) {
    stop = true;
    port.write(1);
  } else {
    stop = false;
    loop();
    port.write(0);
  }
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
