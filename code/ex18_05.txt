// p5.js: 가변저항값 2개 수신하여 택시 이동, 택시가 원 안에 들어가면 자전거가 랜덤하게 떨림, 자전거와 만나면 합성음성으로 게임 종료를 알림, 버튼을 누르면 재시작

let speech = new p5.Speech();
let port = new p5.SerialPort();
let x1 = 0, y1 = 0, x2 = 500, y2 = 500;

function setup() {
  createCanvas(500, 500);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  speech.setLang('ko-KR');

  textAlign(CENTER, CENTER);
  textFont('Material Symbols Outlined');
  textSize(50);
}

function draw() {
  background(0);
  stroke(0, 255, 0);
  noFill();
  ellipse(width/2, height/2, 100, 100);
  fill(255, 255, 255);
  let texi = char(unhex('e559'));
  text(texi, x1, y1);
  fill(255, 255, 0);
  let bike = char(unhex('e52f'));
  text(bike, x2, y2);
  
  chase();
  gameover();
}

function chase() {
  let dx = x1 - x2;
  let dy = y1 - y2;
  d = dist(width/2, height/2, x1, y1);
  if (d < 50) {
    dx = int(random(3)) - 1;
    dy = int(random(3)) - 1;
  }
  if (dx > 0) x2++;
  if (dx < 0) x2--;
  if (dy > 0) y2++;
  if (dy < 0) y2--;
}

function gameover() {
  let d = dist(x1, y1, x2, y2);
  if (d < 5) {
    text("게임 종료", width/2, height/2);
    speech.speak('게임 종료');
    noLoop();
  }
}

function begin() {
  x2 = 500;
  y2 = 500;
  loop();
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write('A');
}

function serialEvent() {
  let str = port.readLine();
  if (!str) return;
  trim(str);
  let value = split(str, ',');
  x1 = map(float(value[0]), 0, 1023, 0, width);
  y1 = map(float(value[1]), 0, 1023, 0, height);
  if(int(value[2]) === 0) begin();
  port.write('B');
}
