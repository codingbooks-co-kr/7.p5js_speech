// p5.js: 버튼값 수신하여 아이콘명을 음성 출력

let speech = new p5.Speech();
let port = new p5.SerialPort();
let words = ['자전거', '버스', '비행기', '보트', '택시', '지하철'];
let traffic = ['eb29', 'e530', 'e539', 'e532', 'e559', 'e56f'];
let index = 0, value = 0, prevValue = 0, x = 0, y = 0, tx = 0.0, ty = 0.5;

function setup() {
  createCanvas(500, 400);
  textFont('Material Symbols Outlined');
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
  speech.speak(words[index]);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('아이콘: 자전거, 버스, 비행기, 보트, 택시, 지하철', 20, 30);
  text('[버튼을 누르면 아이콘 랜덤 출현]', 20, 60);

  textSize(100);
  let icon = char(unhex(traffic[index]));
  text(icon, x, y);

  x = noise(tx) * width;
  y = noise(ty) * height;
  tx += 0.01;
  ty += 0.01;
  if (value === 0 && prevValue === 1) {
    index = int(random(traffic.length));
    speech.speak(words[index]);
  }
  prevValue = value;
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write('A');
}

function serialEvent() {
  value = port.read();
  port.write('B');
}
