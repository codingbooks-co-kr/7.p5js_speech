// 음성인식으로 동물 이모지를 랜덤하게 출력

let rec = new p5.SpeechRec('ko-KR', recResult);
let animals = ['🐵', '🐱', '🐯', '🐷', '🐪', '🐘'];
let index = 0, stop = false;

function setup() {
  createCanvas(400, 200);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  textAlign(CENTER, CENTER);
  text('인식단어: 랜덤, 정지', width/2, 30);
  textSize(100);
  text(animals[index], width/2, height/2);

  if (stop) {
    noLoop();
  } else {
    index = int(random(animals.length));
  }
}

function recResult() {
  let result = rec.resultString;
  console.log(result);
  if (result.indexOf('정지') !== -1) { 
    stop = true;
  } else if (result.indexOf('랜덤') !== -1) {
    stop = false;
    loop();
  }
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
