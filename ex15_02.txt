// p5.js: 음성 제어(→정지, 일, 이, 삼, 사)로 정수(→0,1,2,3,4) 송신

let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let result = '', str = '', value = 0;

function setup() {
  createCanvas(500, 200);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 일(1), 이(2), 삼(3), 사(4), 정지(중지)', 20, 40);
  textSize(50);
  text('인식결과: ' + str, 20, 120);

  if (result.indexOf('일') !== -1 || result.indexOf('1') !== -1) { 
    str = '일(1)';
    value = 1;
  } else if (result.indexOf('이') !== -1 || result.indexOf('2') !== -1) {
    str = '이(2)';
    value = 2;
  } else if (result.indexOf('삼') !== -1 || result.indexOf('3') !== -1) {
    str = '삼(3)';
    value = 3;
  } else if (result.indexOf('사') !== -1 || result.indexOf('4') !== -1) {
    str = '사(4)';
    value = 4;
  } else if (result.indexOf('정지') !== -1 || result.indexOf('중지') !== -1) { 
    str = '정지(중지)';
    value = 0;
  }
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write(0);
}

function serialEvent() {
  let signal = port.read();
  port.write(value);
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
