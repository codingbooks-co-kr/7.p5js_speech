// 음성 제어(→랜덤, 정지)로 동물 이모지를 랜덤 선택하고 음성합성으로 이모지명 출력

let speech = new p5.Speech();
let rec = new p5.SpeechRec('ko-KR', recResult);
let words = ['원숭이', '고양이', '호랑이', '돼지', '낙타', '코끼리'];
let animals = ['🐵', '🐱', '🐯', '🐷', '🐪', '🐘'];
let index = 0, stop = false, toggle = true;

function setup() {
  createCanvas(400, 200);
  speech.setLang('ko-KR');
  speech.started(speechStart);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  textAlign(CENTER, CENTER);
  text('인식단어: 랜덤, 정지', width/2, 30);
  textSize(100);
  text(animals[index], width/2, height/2);

  if (stop) {
    noLoop();
  } else {
    index = int(random(animals.length));
  }
}

function recResult() {
  let result = rec.resultString;
  console.log(result);
  if (result.indexOf('정지') !== -1) {
    stop = true;
    if (toggle) {
      speech.speak(words[index]);
      toggle = false;
    }
  } else if (result.indexOf('랜덤') !== -1) {
    stop = false;
    toggle = true;
    loop();
  }
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}

function speechStart() {
  textSize(20);
  text(words[index], width/2, 160);
}
