// 음성 제어(→자전거, 버스, 비행기, 보트, 택시, 지하철)로 아이콘 표시, 좌우 반사, 부저음 출력

let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let words = ['자전거', '버스', '비행기', '보트', '택시', '지하철'];
let traffic = ['eb29', 'e530', 'e539', 'e532', 'e559', 'e56f'];
let index = 0, x = 250, dx = 3, value = false, prevValue = false;

function setup() {
  createCanvas(500, 200);
  textFont('Material Symbols Outlined');
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  rec.onEnd = recEnd;
  rec.start(true, false);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 자전거, 버스, 비행기, 보트, 택시, 지하철', 20, 30);
  text('인식결과: ' + words[index], 20, 60);

  textSize(100);
  let icon = char(unhex(traffic[index]));
  text(icon, x, 150);
  x += dx;
  if (x < 0 || x > width) {
    dx *= -1;
    value = !value;
  }
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write('A');
}

function serialEvent() {
  let signal = port.read();
  if (value === true && prevValue  === false) {
    port.write(0);
  } else if (value == false && prevValue === true) {
    port.write(1);
  } else {
    port.write(2);
  }
  prevValue = value;
}

function recResult() {
  let result = rec.resultString;
  console.log(result);
  if (result.indexOf('자전거') !== -1) {
    index = 0;
  } else if (result.indexOf('버스') !== -1) {
    index = 1;
  } else if (result.indexOf('비행기') !== -1) { 
    index = 2;
  } else if (result.indexOf('보트') !== -1) {
    index = 3;
  } else if (result.indexOf('택시') !== -1) {
    index = 4;
  } else if (result.indexOf('지하철') !== -1) {
    index = 5;
  }
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
