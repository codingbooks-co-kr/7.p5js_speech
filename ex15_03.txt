// p5.js: 음성 제어(→오른쪽, 왼쪽, 정지)로 이동하는 선의 x축 위치값(1바이트:0∼255) 송신

let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let result = '', value = 0, x = 200, dx = 0;

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(value);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 오른쪽, 왼쪽, 정지', 20, 30);
  text('LED 밝기: ' + nf(value, 3), 130, 100);
  
  stroke(255, 0, 0);
  strokeWeight(2);
  line(x, 0, x, height);
  x += dx;
  x = constrain(x, 0, width);

  if (result.indexOf('오른쪽')!==-1) {
    dx = 1;
  } else if (result.indexOf('왼쪽')!==-1) {
    dx = -1;
  } else if (result.indexOf('정지')!==-1) {
    dx = 0;
  }
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write('A');
}

function serialEvent() {
  let signal = port.read();
  value = int(map(x, 0, width, 0, 255));
  port.write(value);  
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
