// 음성인식 후 인식 결과를 합성음성으로 출력 (시간 초과로 음성인식 종료 시 자동 재시작)

let speech = new p5.Speech();
let rec = new p5.SpeechRec('ko-KR', recResult);
let speaking = false;

function setup() {
  noCanvas();
  speech.setLang('ko-KR');
  speech.started(speechStart);
  speech.ended(speechEnd);
  rec.onEnd = recEnd;
  rec.start(true, false);
  createP('[음성인식 시작]');
}

function recResult() {
  createDiv(rec.resultString);
  speech.speak(rec.resultString);
}

function speechStart() {
  console.log('음성합성 시작');
  speaking = true;
  rec.stop();
}

function speechEnd() {
  console.log('음성합성 종료');
  speaking = false;
  rec.start();
}

function recEnd() {
  if (!speaking) {
    console.log('음성인식 재시작');
    rec.start();
  }
}
