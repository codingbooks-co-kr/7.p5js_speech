// p5.js: 음성 제어(→온/오프 또는 점등/소등)로 정수(0 또는 1) 송신

let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let result = '', value = 0;

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 온(점등), 오프(소등)', 20, 40);

  if (result.indexOf('온') !== -1 || result.indexOf('점등') !== -1) { 
    fill(255, 0, 0);
    value = 1;
  } else if (result.indexOf('오프') !== -1 || result.indexOf('소등') !== -1) { 
    fill(255, 255, 255);
    value = 0;
  }
  rect(100, 80, 200, 80);
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write(0);
}

function serialEvent() {
  let signal = port.read();
  port.write(value); 
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
