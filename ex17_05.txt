// P5.js: 초음파센서로 측정한 거리가 가까워지면 동물 이모지와 이모지명을 랜덤 출력

let speech = new p5.Speech();
let port = new p5.SerialPort();
let words = ['원숭이', '고양이', '호랑이', '돼지', '낙타', '코끼리'];
let animals = ['🐵', '🐱', '🐯', '🐷', '🐪', '🐘'];
let index = 0, value = 0, stop = false, toggle = true;

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
}

function draw() {
  background(0);
  fill(255);
  textSize(100);
  textAlign(CENTER, CENTER);
  text(animals[index], width/2, height/2);

  if (stop) {
    noLoop();
  } else {
    index = int(random(animals.length));
  }
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write('A');
}

function serialEvent() {
  value = port.read();
  if (value < 20) {
    stop = true;
    if (toggle) {
      speech.speak(words[index]);
      toggle = false;
    }
  } else {
    stop = false;
    toggle = true;
    loop();
  }
  port.write('B');
}
