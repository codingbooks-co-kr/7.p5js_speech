// 마우스를 누를 때만 음성인식 실행

// 음성인식 객체 생성 및 초기화 ❶한국어 설정, ❷음성인식 결과 반환 시 recResult() 호출
let rec = new p5.SpeechRec('ko-KR', recResult);

function setup() {
  noCanvas();
  createP('[음성인식 시작]');
  rec.onEnd = recEnd;
  rec.start(false, false);
}

function recResult() {
  createDiv(rec.resultString);
}

function recEnd() {
  createDiv('[마우스를 누르면 음성인식 재시작]');
}

function mousePressed() {
  rec.start();
}
