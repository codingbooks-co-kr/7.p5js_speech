// 운송 수단을 말하면 아이콘으로 출력

let rec = new p5.SpeechRec('ko-KR', recResult);
let traffic = ['eb29', 'e530', 'e539'];
let icon, index = -1;

function setup() {
  createCanvas(400, 200);
  textFont('Material Symbols Outlined');
  textAlign(CENTER, CENTER);
  createP('인식단어: 자전거, 버스, 비행기');

  rec.onEnd = recEnd;
  rec.start(true, false);
}

function draw() {
  background(200);
  if (index >=0 && index < traffic.length) {
    textSize(200);
    icon = char(unhex(traffic[index]));	
    text(icon, width/2, height/2);
  } else {
    textSize(40);
    text('운송수단을 말하세요', width/2, height/2);
  }
}

function recResult() {
  let result = rec.resultString;
  console.log(result);
  if (result.indexOf('자전거') !== -1) {
    index = 0;
  } else if (result.indexOf('버스') !== -1) {
    index = 1;
  } else if (result.indexOf('비행기') !== -1) { 
    index = 2;
  } else {
    index = -1;
  }
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
