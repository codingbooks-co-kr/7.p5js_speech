// 음성으로 좌우 이동하는 선의 현재 위치에 원 대신 사각형으로 변경

let rec = new p5.SpeechRec('ko-KR', recResult);
let move_x = 200, dx = 0;

function setup() {
  createCanvas(400, 200);
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 왼쪽, 오른쪽, 정지', 20, 30);
  
  stroke(255);
  line(move_x, 0, move_x, height);
  
  for (let x = 0; x <= width; x += 50) {
    if (move_x > x-25 && move_x < x+25) {
      fill(255, 0, 0);
      rectMode(CENTER);
      rect(x, height/2, 50, 50);
    } else {
      fill(255); 
      ellipse(x, height/2, 50, 50);
    }
  }
  move_x += dx;
  if (move_x < 0) move_x = width;
  if (move_x > width) move_x = 0;
}

function recResult() {
  let result = rec.resultString;
  console.log(result);
  if (result.indexOf('왼쪽') !== -1) {
    dx = -1;
  } else if (result.indexOf('오른쪽') !== -1) {
    dx = 1;
  } else if (result.indexOf('정지') !== -1) {
    dx = 0;
  }
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
