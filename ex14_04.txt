// p5.js: 가변저항값 수신하여 원을 좌우 이동하고 원의 위치를 음성으로 출력2

let speech = new p5.Speech();
let port = new p5.SerialPort();
let value = 0, newValue = 0, num = 0, prevNum = 0;

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('가변저항: ' + nf(value, 4), 20, 40);
  stroke(255, 255, 0);
  line(width/2, 0, width/2, height);
  ellipse(newValue, height/2, 50, 50);
}

function mousePressed() {
  speech.speak(`가변저항값은 ${value}입니다`);
}

function serialEvent() {
  let str = port.readLine();
  if (!str) return;
  trim(str);
  value = int(str);
  newValue = map(value, 0, 1023, 0, width-1);
  
  num = int(newValue / (width/2));
  if (num != prevNum) {
    if (num === 0) {
      speech.speak('왼쪽');
    } else if (num === 1) {
      speech.speak('오른쪽');
    }
  }
  prevNum = num;
}
