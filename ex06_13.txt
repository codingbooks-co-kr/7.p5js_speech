// 1000개의 점의 상하좌우 및 랜덤 이동을 음성으로 제어

let rec = new p5.SpeechRec('ko-KR', recResult);
let x = [], y = [], c = [];
let dx = 0, dy = 0, num = 1000;

function setup() {
  createCanvas(400, 400);
  for (let i = 0; i < num; i++) {
    // 배열 요소의 초깃값 설정
    x[i] = random(width);
    y[i] = random(height);
    c[i] = color(random(255), random(255), random(255));
  }
  createP('인식단어: 왼쪽, 오른쪽, 위쪽, 아래쪽, 랜덤, 정지');
  rec.onEnd = recEnd; 
  rec.start(true, false); 
}

function draw() {
  // 페이딩 효과
  noStroke();
  fill(0, 10);
  rect(0, 0, width, height);

  for (let i = 0; i < num; i++) {
    stroke(c[i]);
    strokeWeight(5);
    point(x[i], y[i]);
    x[i] += random(dx);
    y[i] += random(dy);

    if (x[i] < 0) x[i] = width;
    if (x[i] > width) x[i] = 0;
    if (y[i] < 0) y[i] = height;
    if (y[i] > height) y[i] = 0;
  }
}

function recResult() {
  let result = rec.resultString;
  console.log(result);
  if (result.indexOf('왼쪽') !== -1) {
    dx = -1;
    dy = 0;
  } else if (result.indexOf('오른쪽') !== -1) {
    dx = 1;
    dy = 0;
  } else if (result.indexOf('위쪽') !== -1) {
    dx = 0;
    dy = -1;
  } else if (result.indexOf('아래쪽') !== -1) {
    dx = 0;
    dy = 1;
  } else if (result.indexOf('랜덤') !== -1) {
    dx = random(-1, 1);
    dy = random(-1, 1);
  } else if (result.indexOf('정지') !== -1) {
    dx = 0;
    dy = 0;
  }
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
