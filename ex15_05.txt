// p5.js: 음성 제어로 0 또는 1 송신, 버튼값 수신하여 텍스트 변경 및 음성 출력

let speech = new p5.Speech();
let rec = new p5.SpeechRec('ko-KR', recResult);
let port = new p5.SerialPort();
let result = '', value1 = 0, value2 = 0, prevValue = 0;
let led = 'LED OFF', button = 'BUTTON OFF';

function setup() {
  createCanvas(400, 200);
  port.open('COM5');
  port.on('open', serialOpen);
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
  rec.onEnd = recEnd;
  rec.start(true, true);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 온(점등), 오프(소등)', 20, 40);
  
  fill(0, 255, 0);
  textSize(50);
  text(led, 100, 100);
  text(button, 50, 150);

  if (result.indexOf('온') !== -1 || result.indexOf('점등') !== -1) {
    led = 'LED ON';
    value1 = 1;
  } else if (result.indexOf('오프') !== -1 || result.indexOf('소등') !== -1) { 
    led = 'LED OFF';
    value1 = 0;
  }

  if (value2 === 0) {
    button = 'BUTTON ON';
    if (prevValue === 1) {
      speech.speak('버튼을 눌렀어요');
    }
  } else if (value2 === 1) {
    button = 'BUTTON OFF';
  }
  prevValue = value2;
}

function serialOpen() {
  console.log('Start!');
  port.clear();
  port.write(0);
}

function serialEvent() {
  value2 = port.read();
  port.write(value1);
}

function recResult() {
  result = rec.resultString;
  console.log(result);
}

function recEnd() {
  console.log('[음성인식 재시작]');
  rec.start();
}
