// 마우스를 누를 때마다 배열의 문장을 차례로 합성음성으로 출력2

let speech = new p5.Speech();
let words = ['안녕하세요', '반가워요', '고마워요'];
let index = 0;

function setup() {
  createCanvas(400, 200);
  speech.setLang('ko-KR');
  speech.setVoice('Google 한국의');
  speech.started(speechStart);
  speech.ended(speechEnd);

  background(0);
  fill(255, 255, 0);
  textAlign(CENTER, CENTER);
  textSize(30);
  text('마우스를 누르세요!', width/2, height/2);
}

function speechStart() {
  background(random(255), random(255), random(255));
  text(words[index], width/2, height/2);
  index = (index+1) % words.length;
}

function speechEnd() {
  background(0);
  text('마우스를 누르세요!', width/2, height/2);
}

function mousePressed() {
  speech.speak(words[index]);
}
