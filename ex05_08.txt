// 버튼을 누를 때마다 음성합성으로 한국어 문장 출력

let speech = new p5.Speech();
let myInput, button, words = '안녕하세요';

function setup() {
  createCanvas(400, 200);
  speech.setLang('ko-KR');
  speech.setVoice('Google 한국의');

  background(0);
  myInput = createInput(words);
  myInput.position(20, 80);
  myInput.size(200);

  button = createButton('Text To Speech');
  button.position(250, 80);
  button.mousePressed(textToSpeech);
}

function textToSpeech() {
  background(random(255), random(255), random(255));
  speech.speak(myInput.value());
}
